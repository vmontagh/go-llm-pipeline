	ID string
	Payload io.Reader
	Metadata map[string]any
}

func AssetFromReader(id string, r io.Reader, metadata map[string]any) Asset

func AssetFromString(id, s string, metadata map[string]any) Asset

	// contains filtered or unexported fields
}

func NewBuilder[T any, R any](p provider.Provider[T], parser func(string) (R, error)) *Builder[T, R]

func NewTextBuilder[C any](p provider.Provider[C]) *Builder[C, string]

func (b *Builder[T, R]) Build() (*Template[T, R], error)

func (b *Builder[T, R]) WithBackOff(bo backoff.BackOff) *Builder[T, R]

func (b *Builder[T, R]) WithLogger(l *slog.Logger) *Builder[T, R]

type ExtractOption interface {
	// contains filtered or unexported methods
}

func WithFeedbackFormatter(fn func(FeedbackInput) string) ExtractOption

func WithMaxAttempts(n int) ExtractOption

func WithSchema(schema map[string]any) ExtractOption

func WithStrictDecoding(strict bool) ExtractOption

type ExtractRequest struct {
	Prompt string
	ImageData []byte
	JobID string
	Publish PublishFn
}

type FeedbackInput struct {
	Schema map[string]any
	Raw string
	ParseError error
	Fields []*FieldError
}

type FieldError struct {
	Path string
	Rule string
	Detail string
}

func (e *FieldError) Error() string

func (e *FieldError) Unwrap() error

type PublishFn = func(eventType, itemID, message string, isError bool)

type Template[T any, R any] struct {
	// contains filtered or unexported fields
}

func (t *Template[T, R]) Analyze(
	ctx context.Context,
	prompt string,
	imageData []byte,
	jobID string,
	publish func(eventType, itemID, message string, isError bool),
) (R, error)

func (t *Template[T, R]) AnalyzeAsset(
	ctx context.Context,
	prompt string,
	asset Asset,
	publish func(eventType, itemID, message string, isError bool),
) (R, error)

func (t *Template[T, R]) AnalyzeMessages(
	ctx context.Context,
	msgs []provider.Message,
	jobID string,
	publish func(eventType, itemID, message string, isError bool),
) (R, error)

func (t *Template[T, R]) AnalyzeWithTools(ctx context.Context, msgs []provider.Message, loop ToolLoop, jobID string, publish PublishFn) (R, error)

func (t *Template[T, R]) Name() string

func (t *Template[T, R]) Provider() provider.Provider[T]

func (t *Template[T, R]) StreamMessages(
	ctx context.Context,
	msgs []provider.Message,
	jobID string,
	publish func(eventType, itemID, message string, isError bool),
) iter.Seq2[provider.Chunk, error]

type TextAnalyzer interface {
	Analyze(ctx context.Context, prompt string, imageData []byte, jobID string, publish PublishFn) (string, error)
}

type Tool struct {
	// contains filtered or unexported fields
}

func NewTool[A, R any](name, description string, fn func(context.Context, A) (R, error)) (Tool, error)

type ToolLoop struct {
	Tools []Tool
	MaxSteps int
	ReportInvalidCalls bool
}

type Tracer interface {
	StartRegion(ctx context.Context, name string) interface{ End() }
}
