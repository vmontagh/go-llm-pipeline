package provider

import (
	context "context"
	errors "errors"
	iter "iter"
	time "time"
)

const (
	MetricGuardWaits          = "guard_waits"
	MetricGuardLimitDecreases = "guard_limit_decreases"
	MetricGuardBreakerOpens   = "guard_breaker_opens"
)

const DefaultMaxTokens = 500

var (
	ErrMissingAPIKey = errors.New("missing API key")
	ErrAllKeysCoolingDown = errors.New("all API keys are in cooldown")
	ErrQuotaExceeded = errors.New("quota exceeded")
	ErrRateLimitExceeded = errors.New("rate limit exceeded")
	ErrEmptyCandidate = errors.New("empty candidate returned")
	ErrInvalidResponsePart = errors.New("invalid response part returned")
	ErrUnexpectedStatus = errors.New("unexpected status code returned")
)

var (
	ErrStreamUnsupported = errors.New("provider does not support streaming")
	ErrIncompleteStream = errors.New("stream ended before the reply was done")
)

var ErrInvalidGuardConfig = errors.New("invalid guard config")

var ErrInvalidMessages = errors.New("invalid messages")

var ErrMessagesUnsupported = errors.New("provider does not support messages")

var ErrToolsUnsupported = errors.New("provider does not support tools")

func DefaultEnrichError(_ ErrorTier, rawErr error, _ []byte) error

func HasTools(msgs []Message) bool

func ValidateMessages(msgs []Message) error

type Adder interface {
	Add(name string, delta int64)
}

type BreakerState int

const (
	BreakerUnknown BreakerState = iota
	BreakerClosed
	BreakerOpen
	BreakerHalfOpen
)

func (s BreakerState) String() string

type CallOutcome int

const (
	OutcomeSuccess CallOutcome = iota
	OutcomeProviderError
	OutcomeTerminal
)

type CallResult struct {
	Text string
	Err error
	Outcome CallOutcome
	StatusCode int
	Body []byte
	RetryAfter time.Duration
	ToolCalls []ToolCall
	Parts []Part
}

type Chunk struct {
	Text string
	Usage *Usage
}

type ClassifyInput struct {
	Err        error
	StatusCode int
	Body       []byte
	RetryAfter time.Duration
}

type ErrorTier int

const (
	TierTerminal ErrorTier = iota
	TierShortTransient
	TierShortRateLimit
	TierLongTransientBounded
	TierLongTransientInfinite
)

type Guard[T any] struct {
	Provider[T]
	// contains filtered or unexported fields
}

func NewGuard[T any](p Provider[T], opts ...GuardOption) (*Guard[T], error)

func (g *Guard[T]) Call(ctx context.Context, lease *Lease[T], prompt string, imageData []byte) CallResult

func (g *Guard[T]) CallMessages(ctx context.Context, lease *Lease[T], msgs []Message) CallResult

func (g *Guard[T]) CallTools(ctx context.Context, lease *Lease[T], msgs []Message, tools []ToolSpec) CallResult

func (g *Guard[T]) Limit() int

func (g *Guard[T]) State() BreakerState

func (g *Guard[T]) Stream(ctx context.Context, lease *Lease[T], msgs []Message) (iter.Seq2[Chunk, error], CallResult)

type GuardClock interface {
	Now() time.Time
	After(d time.Duration) <-chan time.Time
}

type GuardOption func(*guardConfig)

func WithBreaker(failures int, openFor time.Duration) GuardOption

func WithConcurrencyLimits(minLimit, maxLimit int) GuardOption

func WithGuardClock(clock GuardClock) GuardOption

func WithGuardMetrics(a Adder) GuardOption

func WithLatencyTarget(d time.Duration) GuardOption

type KeyPool[T any] struct {
	// contains filtered or unexported fields
}

func NewKeyPool[T any](items []T) *KeyPool[T]

func NewKeyPoolWithClock[T any](items []T, clk PoolClock) *KeyPool[T]

func (p *KeyPool[T]) Acquire(ctx context.Context) (*Lease[T], error)

func (p *KeyPool[T]) Available() int

func (p *KeyPool[T]) Len() int

func (p *KeyPool[T]) MinCooldown() time.Duration

type Lease[T any] struct {
	Value T
	Index int
	// contains filtered or unexported fields
}

func (l *Lease[T]) MarkCooldown(policy TierPolicy, retryAfter time.Duration)

func (l *Lease[T]) Release()

type Message struct {
	Role  Role
	Parts []Part
}

func TextMessage(role Role, text string) Message

type MessageCaller[T any] interface {
	CallMessages(ctx context.Context, lease *Lease[T], msgs []Message) CallResult
}

func AsMessageCaller[T any](p Provider[T]) (MessageCaller[T], bool)

type Part struct {
	Text string
	Image []byte
	ToolCall *ToolCall
	ToolResult *ToolResult
	Signature string
}

type PoolClock interface {
	Now() time.Time
}

type Provider[T any] interface {
	Name() string
	Model() string
	ProviderLabel() string
	TraceRegionName() string
	Keys() *KeyPool[T]
	ClassifyError(ClassifyInput, RetryContext) ErrorTier
	EnrichError(tier ErrorTier, rawErr error, body []byte) error
	Call(ctx context.Context, lease *Lease[T], prompt string, imageData []byte) CallResult
}

type RealPoolClock struct{}

func (RealPoolClock) Now() time.Time

type RetryContext struct {
	Attempt int
	Elapsed time.Duration
}

type Role int

const (
	RoleUnknown Role = iota
	RoleSystem
	RoleUser
	RoleAssistant
	RoleTool
)

func (r Role) String() string

type Streamer[T any] interface {
	Stream(ctx context.Context, lease *Lease[T], msgs []Message) (iter.Seq2[Chunk, error], CallResult)
}

func AsStreamer[T any](p Provider[T]) (Streamer[T], bool)

type TierPolicy struct {
	Cooldown time.Duration
	HonorRetryAfter bool
	L2MaxRetries int
	L2BaseBackoff time.Duration
	L2MaxBackoff  time.Duration
	EscalateAfterAttempts int
	EscalateAfter         time.Duration
}

type ToolCall struct {
	ID   string
	Name string
	Arguments string
}

type ToolCaller[T any] interface {
	CallTools(ctx context.Context, lease *Lease[T], msgs []Message, tools []ToolSpec) CallResult
}

func AsToolCaller[T any](p Provider[T]) (ToolCaller[T], bool)

type ToolResult struct {
	CallID  string
	Name    string
	Content string
}

type ToolSpec struct {
	Name        string
	Description string
	Parameters map[string]any
}

type Usage struct {
	Input  int
	Output int
}
